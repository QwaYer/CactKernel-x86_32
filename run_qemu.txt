qemu-system-i386 \
    -accel kvm \
    -cpu host \
    -m 4G \
    -cdrom build/cact.iso \
    -boot d \
    -serial stdio \
    -rtc base=localtime \
    -M q35 \
    -vga std \
    -display gtk \
    -drive file=build/nvme.img,if=none,id=nvm0,format=raw \
    -d guest_errors -D qemu.log \
    -device nvme,serial=cact-1337,drive=nvm0 \
    -nic user,model=virtio-net-pci \
    -device qemu-xhci \
    -no-reboot \
    -no-shutdown


# Образ диска: надёжнее всего собрать скриптом (userspace + ext4 + правильные пути):
#   ./build_disk.sh
#
# Важно: на ext4-образе init должен быть в КОРНЕ файловой системы как /bin/init
# (alias binfs в ядре смотрит туда). Путь :/raw/bin/... — ошибка: ядро не найдёт init.
#
# Ручная сборка (из каталога CactKernel-x86_32), если нужно без пересборки ELF:
#   mkdir -p build
#   qemu-img create -f raw build/nvme.img 512M
#   mkfs.ext4 -q -b 4096 -I 128 -O ^64bit,^metadata_csum,^has_journal build/nvme.img
#   e2mkdir build/nvme.img:/bin
#   e2cp ../Cgoct-x86_32/cgoct     build/nvme.img:/bin/init
#   e2cp ../Cactsole-x86_32/cactsole build/nvme.img:/bin/cactsole
